【1】串口数据采集 (serial_reader.py)
----------------------------------------------------------------
# 默认采集 COM24 (无线端口)
python scripts\serial_reader.py

# 指定端口
python scripts\serial_reader.py -p COM24

# 持续采集 60 秒
python scripts\serial_reader.py -t 60

# 保存数据到 data/ 目录
python scripts\serial_reader.py -t 60 --save

# 指定输出文件名
python scripts\serial_reader.py -t 60 --save -o data\my_test.txt

# 查看可用串口
python scripts\serial_reader.py --list

# 采集指定行数
python scripts\serial_reader.py -n 1000

【8】Git 命令速查
----------------------------------------------------------------
# 查看工作区状态
git st

# 查看完整状态信息
git status


# 查看最近提交历史
git last

# 查看所有分支列表
git br

# 切换到 main 分支
git switch main

# 创建并切换新分支
git switch -c fix/mag_yaw

# 添加所有更改
git add .

# 添加指定文件
git add code\ins\imu660.c

# 取消暂存指定文件
git unstage code\ins\imu660.c

# 提交更改
git commit -m "fix: 修复磁力计航向融合问题"

# 推送到远程分支 main
git push -u origin main

# 查看远程仓库
git remote -v

# 设置远程仓库地址
git remote set-url origin https://github.com/Aorcheved001/race.git

# 查看 Git 代理设置
git config --global --get http.proxy
git config --global --get https.proxy

# 设置 Git 使用 Clash 代理 (解决 GitHub 连接问题)
git config --global http.proxy http://127.0.0.1:7890
git config --global https.proxy http://127.0.0.1:7890

# 取消 Git 代理
git config --global --unset http.proxy
git config --global --unset https.proxy

# 测试 GitHub 连接是否正常
git ls-remote origin

# 同步远程仓库 (安全方式，不会自动合并)
git fetch origin
git switch main
git pull --ff-only origin main

# 查看本地与远程的差异
git fetch origin
git lg

# 如果本地分支(如main、fix/mag_yaw)落后于远程分支，先暂存本地更改再拉取：
# 步骤：查看状态 -> 备份更改 -> 拉取远程
# 警告：使用 git reset --hard 会丢弃所有未提交的更改，请谨慎操作！


================================================================
  常用工作流程
================================================================

# 标准开发流程: 编码 -> 同步 -> 编译 -> 测试
python tools\sync.py --force
tools\headless_build.bat --flash-only
python scripts\serial_reader.py -t 60 --save

# 磁力计校准流程: 编译 -> 校准 GUI -> 【8】Git操作 -> 推送更改
tools\headless_build.bat --flash-only
python mag_calibration\plotter.py

# 推送更改到 GitHub: 查看状态 -> 暂存 -> 提交 -> 推送
git st
git add .
git commit -m "feat: 添加新功能"
git push -u origin main

# 从远程拉取最新代码: 查看状态 -> 切换到 main
git st
git fetch origin
git switch main
git pull --ff-only origin main

# 创建功能分支: 查看状态 -> 创建 -> 提交 -> 推送
git switch -c feat/my_change
git add .
git commit -m "feat: 新功能描述"
git push

# GitHub 连接失败解决方案: 开启 Clash -> 设置代理 -> 测试连接 -> 推送
git config --global http.proxy http://127.0.0.1:7890
git config --global https.proxy http://127.0.0.1:7890
git ls-remote origin
git push -u origin main

# 漂移测试流程: 编译 -> 采集60秒 -> 分析
tools\headless_build.bat --flash-only
python scripts\serial_reader.py -t 60 --save -o data\drift_test.txt
python scripts\analyze_drift.py data\drift_test.txt
